cmake_minimum_required(VERSION 3.24)

# 1. Project name and version
project(mino_project VERSION 1.0.0 LANGUAGES CXX)

# 2. C++ Standard settings
if(NOT DEFINED CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17) # Default to C++17 if not specified
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# 3. Cross-platform path and installation layout
# Force libraries to install into "lib" (override distro default like "lib64")
# NOTE: this must be set before including GNUInstallDirs so subprojects see the value.
set(CMAKE_INSTALL_LIBDIR "lib" CACHE STRING "Installation directory for libraries" FORCE)
# GNUInstallDirs provides standard install directories across platforms (bin, lib, include, etc.)
include(GNUInstallDirs)

# Define standard package config installation destination
set(MINO_INSTALL_CMAKEDIR "${CMAKE_INSTALL_LIBDIR}/cmake/mino")

# 4. OS-specific configuration
if(WIN32)
  # Prevent Windows headers from defining min/max macros (MSVC)
  add_compile_definitions(NOMINMAX)
elseif(APPLE)
  # macOS RPATH settings for dynamic libraries
  set(CMAKE_MACOSX_RPATH ON)
  set(CMAKE_INSTALL_RPATH "@loader_path/../${CMAKE_INSTALL_LIBDIR}")
elseif(UNIX)
  # Linux / UNIX RPATH settings
  set(CMAKE_INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
endif()

# 5. Output directories for build targets
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin) # Executables and Windows DLLs
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib) # Static libraries and Windows import libs (.lib)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib) # Shared libraries (.so, .dylib)

# 6. Subproject directories
add_subdirectory(core) # No dependencies

add_subdirectory(external) # Dependency: core

add_subdirectory(network/mino_network)         # Dependency: core
add_subdirectory(network/mino_network_libssh2) # Dependencies: core, network
add_subdirectory(network/mino_network_curl)    # Dependencies: core, network
add_subdirectory(network/mino_network_openssl) # Dependencies: core, network

# 7. Include CMake module for package configuration helpers
include(CMakePackageConfigHelpers)

# 8. Export the 'minoTargets' bundle generated by submodules to a file
install(
  EXPORT minoTargets
  FILE minoTargets.cmake
  NAMESPACE mino:: # Allows external users to target mino::core, etc.
  DESTINATION ${MINO_INSTALL_CMAKEDIR}
)
# minoTargets.cmake
# - Install destination: <CMAKE_INSTALL_PREFIX>/${MINO_INSTALL_CMAKEDIR}/minoTargets.cmake
# - Role: Contains target metadata exported from submodules (core, network, etc.)
#         via install(TARGETS ... EXPORT minoTargets).
# - Key contents:
#   - Path mappings for built binaries/libraries
#   - Imported target definitions under the 'mino::' namespace (mino::core, mino::network, etc.)
#   - Required compiler options, interface include directories, and link dependencies

# 9. Generate package version file
write_basic_package_version_file(
  ${CMAKE_CURRENT_BINARY_DIR}/minoConfigVersion.cmake
  VERSION ${PROJECT_VERSION}
  COMPATIBILITY SameMajorVersion
)
# minoConfigVersion.cmake
# - Install destination: <CMAKE_INSTALL_PREFIX>/${MINO_INSTALL_CMAKEDIR}/minoConfigVersion.cmake
# - Role: Verifies version compatibility when external projects call find_package(mino <version> REQUIRED).
# - Key contents:
#   - Specifies package version (1.0.0)
#   - SameMajorVersion compatibility rule (e.g., 1.x.x matches 1.0.0, but fails on 2.0.0+ or 0.x)

# 10. Generate Config file for external projects
file(
  WRITE ${CMAKE_CURRENT_BINARY_DIR}/minoConfig.cmake
  "include(CMakeFindDependencyMacro)\n"
  "include(\"\${CMAKE_CURRENT_LIST_DIR}/minoTargets.cmake\")\n"
)
# minoConfig.cmake
# - Install destination: <CMAKE_INSTALL_PREFIX>/${MINO_INSTALL_CMAKEDIR}/minoConfig.cmake
# - Role: Main entry point loaded by CMake when an external project calls find_package(mino).
# - Key contents:
#   - Loads the CMakeFindDependencyMacro module
#   - Includes minoTargets.cmake from the same directory to import targets into the external project scope

# 11. Install Config and Version files
install(
  FILES
    ${CMAKE_CURRENT_BINARY_DIR}/minoConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/minoConfigVersion.cmake
  DESTINATION ${MINO_INSTALL_CMAKEDIR}
)

# 12. Install custom CMake module files (.cmake)
# Installs files under ${CMAKE_SOURCE_DIR}/mino/cmake-module/ to <CMAKE_INSTALL_PREFIX>/${MINO_INSTALL_CMAKEDIR}/
if(EXISTS "${CMAKE_SOURCE_DIR}/mino/cmake-module")
  install(
    DIRECTORY "${CMAKE_SOURCE_DIR}/mino/cmake-module/" # Trailing slash copies directory contents directly
    DESTINATION ${MINO_INSTALL_CMAKEDIR}
    FILES_MATCHING PATTERN "*.cmake"
  )
endif()
